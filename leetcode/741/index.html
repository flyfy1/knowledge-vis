def baseline(grid):
    n = len(grid)
    def paths(r, c, seen):
        if r >= n or c >= n or grid[r][c] == -1:
            return
        seen = seen | {(r, c)}
        if r == c == n - 1:
            yield seen
        else:
            yield from paths(r + 1, c, seen)
            yield from paths(r, c + 1, seen)
    routes = list(paths(0, 0, set()))
    return max((sum(grid[r][c] for r, c in a | b)
                for a in routes for b in routes), default=0)
