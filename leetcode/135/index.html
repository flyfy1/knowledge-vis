def baseline(ratings):
    n = len(ratings)
    candies = [1] * n
    for _ in range(n - 1):
        old = candies[:]
        for i in range(n):
            for j in (i - 1, i + 1):
                if 0 <= j < n and ratings[i] > ratings[j]:
                    candies[i] = max(candies[i], old[j] + 1)
    return sum(candies)
