TURN = {"0": "0", "1": "1", "6": "9", "8": "8", "9": "6"}


def turn(x):
    s = str(x)
    if any(ch not in TURN for ch in s):
        return None  # some digit cannot turn
    # reversed: the last digit comes first; int() drops leading zeros
    return int("".join(TURN[ch] for ch in reversed(s)))


def brute(n):
    count = 0
    for x in range(1, n + 1):
        r = turn(x)
        if r is not None and r != x:
            count += 1
    return count


def turnable_numbers(n):
    stack = [1, 6, 8, 9]
    while stack:
        x = stack.pop()
        if x > n:
            continue
        yield x
        stack.extend(x * 10 + d for d in (0, 1, 6, 8, 9))


def generate_then_check(n):
    return sum(1 for x in turnable_numbers(n) if turn(x) != x)
