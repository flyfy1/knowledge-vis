def skyline_baseline(positions):
    heights = {}
    answer = []
    highest = 0
    for left, side in positions:
        columns = range(left, left + side)
        base = max(heights.get(x, 0) for x in columns)
        top = base + side
        for x in columns:
            heights[x] = top
        highest = max(highest, top)
        answer.append(highest)
    return answer
