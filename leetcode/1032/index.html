class DirectChecker:
    def __init__(self, words):
        self.words = words
        self.limit = max(len(w) for w in words)
        self.stream = ""

    def query(self, letter):
        self.stream = (self.stream + letter)[-self.limit:]
        for word in self.words:
            if self.stream.endswith(word):
                return True
        return False
