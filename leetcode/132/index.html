def baseline(s):
    best = len(s) - 1
    for mask in range(1 << (len(s) - 1)):
        start, parts = 0, []
        for i in range(len(s) - 1):
            if mask & (1 << i):
                parts.append(s[start:i + 1])
                start = i + 1
        parts.append(s[start:])
        if all(p == p[::-1] for p in parts):
            best = min(best, len(parts) - 1)
    return best
