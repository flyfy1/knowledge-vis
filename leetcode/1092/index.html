def minimum_length(a, b):
    m, n = len(a), len(b)

    def visit(i, j):
        if i == m:
            return n - j
        if j == n:
            return m - i
        if a[i] == b[j]:
            return 1 + visit(i + 1, j + 1)
        return 1 + min(visit(i + 1, j), visit(i, j + 1))

    return visit(0, 0)
