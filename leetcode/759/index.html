def brute(schedule):
    busy = [(iv.start, iv.end) for person in schedule for iv in person]
    points = sorted({p for s, e in busy for p in (s, e)})
    free = []
    for a, b in zip(points, points[1:]):          # every piece between endpoints
        if not any(s <= a and b <= e for s, e in busy):   # check every interval
            free.append(Interval(a, b))
    return free
