    SELECT
    a.event_date AS install_dt,
    COUNT(*) AS installs,
    ROUND(SUM(EXISTS (
        SELECT 1 FROM Activity AS b
        WHERE b.player_id = a.player_id
          AND b.event_date = DATE_ADD(a.event_date, INTERVAL 1 DAY)
    )) / COUNT(*), 2) AS Day1_retention
FROM Activity AS a
WHERE a.event_date = (
    SELECT MIN(c.event_date)
    FROM Activity AS c
    WHERE c.player_id = a.player_id
)
GROUP BY a.event_date;
