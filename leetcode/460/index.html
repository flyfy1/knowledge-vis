class Baseline:
    def __init__(self, capacity):
        self.capacity, self.clock, self.items = capacity, 0, {}
    def get(self, key):
        if key not in self.items:
            return -1
        self.clock += 1
        value, freq, _ = self.items[key]
        self.items[key] = (value, freq + 1, self.clock)
        return value
    def put(self, key, value):
        if self.capacity <= 0:
            return
        self.clock += 1
        if key in self.items:
            _, freq, _ = self.items[key]
            self.items[key] = (value, freq + 1, self.clock)
            return
        if len(self.items) == self.capacity:
            victim = min(self.items, key=lambda k: self.items[k][1:])
            del self.items[victim]
        self.items[key] = (value, 1, self.clock)
