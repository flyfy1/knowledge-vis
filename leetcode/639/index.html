from itertools import product

def brute(s):
    def count(d, i=0):
        if i == len(d):
            return 1
        if d[i] == '0':
            return 0
        total = count(d, i + 1)
        if i + 1 < len(d) and int(d[i:i+2]) <= 26:
            total += count(d, i + 2)
        return total

    choices = ['123456789' if c == '*' else c for c in s]
    return sum(count(''.join(d)) for d in product(*choices)) % 1_000_000_007
