def direct(n, lamps, queries):
    active = set(map(tuple, lamps))
    ans = []
    for r, c in queries:
        lit = False
        for x, y in active:
            if x == r or y == c or abs(x-r) == abs(y-c):
                lit = True
        ans.append(int(lit))
        active = {(x, y) for x, y in active
                  if abs(x-r) > 1 or abs(y-c) > 1}
    return ans
