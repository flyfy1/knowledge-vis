def summarize_slow(grid, cells):
    targets, edges = set(), 0
    for r, row in enumerate(grid):
        for c, value in enumerate(row):
            if value == 0:
                hits = sum((r + dr, c + dc) in cells
                           for dr, dc in ((1, 0), (-1, 0),
                                          (0, 1), (0, -1)))
                if hits:
                    targets.add((r, c))
                edges += hits
    return targets, edges
