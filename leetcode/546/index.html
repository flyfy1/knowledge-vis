    def brute(a):
    if not a:
        return 0
    best = 0
    for i in range(len(a)):
        for j in range(i, len(a)):
            if a[j] != a[i]:
                break
            best = max(best, (j-i+1)**2
                       + brute(a[:i] + a[j+1:]))
    return best
