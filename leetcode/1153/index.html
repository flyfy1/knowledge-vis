    from collections import deque

def can_convert_bfs(str1, str2):
    seen = {str1}
    queue = deque([str1])
    while queue:
        s = queue.popleft()
        if s == str2:
            return True
        for c in set(s):                      # a letter that appears
            for d in "abcdefghijklmnopqrstuvwxyz":
                if d != c:
                    t = s.replace(c, d)       # rewrite ALL copies of c
                    if t not in seen:
                        seen.add(t)
                        queue.append(t)
    return False
