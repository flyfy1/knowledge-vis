def full_scan(graph):
    states = [(m, c, t) for m in range(len(graph))
              for c in range(1, len(graph)) for t in (1, 2)]
    win = {s: 1 if s[0] == 0 else 2 if s[0] == s[1] else 0
           for s in states}
    while True:
        new = dict(win)
        for m, c, t in states:
            if win[m, c, t]:
                continue
            if t == 1:
                values = [win[v, c, 2] for v in graph[m]]
            else:
                values = [win[m, v, 1] for v in graph[c] if v != 0]
            if t in values:
                new[m, c, t] = t
            elif values and all(v == 3 - t for v in values):
                new[m, c, t] = 3 - t
        if new == win:
            return win[1, 2, 1]
        win = new
