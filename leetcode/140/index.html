def baseline(s, wordDict):
    words = set(wordDict)
    answer = []
    for mask in range(1 << (len(s) - 1)):
        parts, start = [], 0
        for gap in range(1, len(s)):
            if mask & (1 << (gap - 1)):
                parts.append(s[start:gap])
                start = gap
        parts.append(s[start:])
        if all(part in words for part in parts):
            answer.append(" ".join(parts))
    return answer
