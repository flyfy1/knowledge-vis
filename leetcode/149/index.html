y = x / 2

def baseline(points):
    best = 1
    for i, (x, y) in enumerate(points):
        for j in range(i + 1, len(points)):
            bx, by = points[j]
            dx, dy = bx - x, by - y
            count = sum((px - x) * dy == (py - y) * dx
                        for px, py in points)
            best = max(best, count)
    return best
