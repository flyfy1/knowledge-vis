def baseline(s):
    answer = 0
    for l in range(len(s)):
        counts = {}
        for r in range(l, len(s)):
            ch = s[r]
            counts[ch] = counts.get(ch, 0) + 1
            answer += sum(v == 1 for v in counts.values())
    return answer
