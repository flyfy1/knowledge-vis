def search(piles, k):
    if len(piles) == 1:
        return 0
    best = float("inf")
    for start in range(len(piles) - k + 1):
        payment = sum(piles[start:start + k])
        after = piles[:start] + [payment] + piles[start + k:]
        best = min(best, payment + search(after, k))
    return best  # infinity means impossible; caller returns -1
