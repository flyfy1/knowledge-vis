def baseline(trees):
    if len(trees) == 1:
        return trees[:]
    boundary = set()
    for i, a in enumerate(trees):
        for b in trees[i + 1:]:
            sides = [(b[0]-a[0])*(p[1]-a[1])
                     - (b[1]-a[1])*(p[0]-a[0]) for p in trees]
            if min(sides) >= 0 or max(sides) <= 0:
                boundary.update((tuple(a), tuple(b)))
    return [list(p) for p in boundary]
