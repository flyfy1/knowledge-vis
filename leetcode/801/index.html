def baseline(nums1, nums2):
    n = len(nums1)
    best = n + 1
    for mask in range(1 << n):
        a = [nums2[i] if mask & (1 << i) else nums1[i]
             for i in range(n)]
        b = [nums1[i] if mask & (1 << i) else nums2[i]
             for i in range(n)]
        if all(a[i-1] < a[i] and b[i-1] < b[i]
               for i in range(1, n)):
            best = min(best, mask.bit_count())
    return best
