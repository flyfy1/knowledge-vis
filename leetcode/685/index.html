def baseline(edges):
    n = len(edges)
    for removed in range(n - 1, -1, -1):
        incoming = [0] * (n + 1)
        children = [[] for _ in range(n + 1)]
        for i, (u, v) in enumerate(edges):
            if i != removed:
                incoming[v] += 1
                children[u].append(v)
        roots = [v for v in range(1, n + 1)
                 if incoming[v] == 0]
        if len(roots) != 1 or any(x > 1 for x in incoming):
            continue
        seen, stack = set(), [roots[0]]
        while stack:
            u = stack.pop()
            if u not in seen:
                seen.add(u)
                stack.extend(children[u])
        if len(seen) == n:
            return edges[removed][:]
