import re

def flat(e):                     # no parentheses: number, operator, number, ...
    parts = re.split(r"([+\-*/])", e)
    nums, ops = [int(parts[0])], []
    for o, x in zip(parts[1::2], parts[2::2]):
        if o in "*/":            # * and / at once, left to right
            a, b = nums[-1], int(x)
            nums[-1] = a * b if o == "*" else int(a / b)
        else:
            ops.append(o)
            nums.append(int(x))
    total = nums[0]              # then + and -
    for o, x in zip(ops, nums[1:]):
        total = total + x if o == "+" else total - x
    return total

def brute(s):
    while "(" in s:
        close = s.index(")")                 # the first ")" closes the innermost "("
        start = s.rindex("(", 0, close)
        s = s[:start] + str(flat(s[start + 1:close])) + s[close + 1:]   # rewrite the text
    return flat(s)
