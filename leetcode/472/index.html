def baseline(words):
    dictionary = set(words)
    answer = []
    for word in words:
        n = len(word)
        for mask in range(1, 1 << (n - 1)):
            cuts = [0] + [i for i in range(1, n)
                          if mask & (1 << (i - 1))] + [n]
            if all(word[a:b] in dictionary
                   for a, b in zip(cuts, cuts[1:])):
                answer.append(word)
                break
    return answer
