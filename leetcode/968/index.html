def baseline(nodes, neighbors):
    # neighbors[i] contains parent and immediate children.
    n = len(nodes)
    best = n
    for mask in range(1 << n):
        covered = set()
        for i in range(n):
            if mask & (1 << i):
                covered.add(i)
                covered.update(neighbors[i])
        if len(covered) == n:
            best = min(best, mask.bit_count())
    return best
