from collections import deque

def baseline(begin, end, words):
    queue = deque([[begin]])
    answer, best = [], None
    while queue:
        path = queue.popleft()
        if best is not None and len(path) > best:
            break
        if path[-1] == end:
            best = len(path)
            answer.append(path)
            continue
        for word in words:
            if word not in path and sum(a != b for a, b in zip(path[-1], word)) == 1:
                queue.append(path + [word])
    return answer
