def baseline(k, prices):
    def visit(day, holding, remaining):
        if day == len(prices):
            return float('-inf') if holding else 0
        best = visit(day + 1, holding, remaining)  # rest
        if holding:
            best = max(best, prices[day]
                       + visit(day + 1, False, remaining - 1))
        elif remaining:
            best = max(best, -prices[day]
                       + visit(day + 1, True, remaining))
        return best
    return visit(0, False, k)
