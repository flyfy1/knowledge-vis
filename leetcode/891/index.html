def baseline(nums):
    choices = [[]]
    for value in nums:
        added = []
        for seq in choices:
            added.append(seq + [value])
        choices += added
    total = 0
    for seq in choices:
        if seq:
            total += max(seq) - min(seq)
    return total % 1_000_000_007
