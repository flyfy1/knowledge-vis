    def baseline(stations, k):
    gaps = [b - a for a, b in zip(stations, stations[1:])]

    def best(i, left):
        # smallest worst gap for gaps[i:] using `left` new stations
        if i == len(gaps) - 1:
            return gaps[i] / (left + 1)
        return min(max(gaps[i] / (c + 1), best(i + 1, left - c))
                   for c in range(left + 1))

    return best(0, k)
