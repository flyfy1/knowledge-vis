def baseline(matrix):
    m, n = len(matrix), len(matrix[0])
    def walk(r, c):
        best = 1
        for dr, dc in ((1,0), (-1,0), (0,1), (0,-1)):
            a, b = r + dr, c + dc
            if 0 <= a < m and 0 <= b < n and matrix[a][b] > matrix[r][c]:
                best = max(best, 1 + walk(a, b))
        return best
    return max(walk(r, c) for r in range(m) for c in range(n))
