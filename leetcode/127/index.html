    from collections import deque

def baseline(begin, end, words):
    nodes = list(set(words) | {begin})
    edges = {w: [] for w in nodes}
    for i, a in enumerate(nodes):
        for b in nodes[:i]:
            if sum(x != y for x, y in zip(a, b)) == 1:
                edges[a].append(b)
                edges[b].append(a)
    queue, seen = deque([(begin, 1)]), {begin}
    while queue:
        word, d = queue.popleft()
        if word == end:
            return d
        for neighbor in edges[word]:
            if neighbor not in seen:
                seen.add(neighbor)
                queue.append((neighbor, d + 1))
    return 0
