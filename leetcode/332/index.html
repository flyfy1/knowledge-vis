def baseline(tickets):
    used = [False] * len(tickets)
    best = None
    def visit(route):
        nonlocal best
        if len(route) == len(tickets) + 1:
            if best is None or route < best:
                best = route[:]
            return
        for i, (a, b) in enumerate(tickets):
            if not used[i] and a == route[-1]:
                used[i] = True
                route.append(b)
                visit(route)
                route.pop()
                used[i] = False
    visit(["JFK"])
    return best
