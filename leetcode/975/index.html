def baseline(arr):
    def targets(direction):
        nxt = [-1] * len(arr)
        for i in range(len(arr)):
            candidates = []
            for j in range(i + 1, len(arr)):
                if direction * arr[j] >= direction * arr[i]:
                    candidates.append((direction * arr[j], j))
            if candidates:
                nxt[i] = min(candidates)[1]
        return nxt

    up, down = targets(1), targets(-1)
    n = len(arr)
    odd, even = [False] * n, [False] * n
    odd[-1] = even[-1] = True
    for i in range(n - 2, -1, -1):
        if up[i] != -1:
            odd[i] = even[up[i]]
        if down[i] != -1:
            even[i] = odd[down[i]]
    return sum(odd)
