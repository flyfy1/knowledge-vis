def baseline(matrix, k):
    m, n = len(matrix), len(matrix[0])
    best = float('-inf')
    for top in range(m):
        for bottom in range(top, m):
            for left in range(n):
                for right in range(left, n):
                    total = sum(matrix[r][c]
                                for r in range(top, bottom + 1)
                                for c in range(left, right + 1))
                    if total <= k:
                        best = max(best, total)
    return best
