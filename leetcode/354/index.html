def baseline(envelopes):
    ordered = sorted(envelopes)
    dp = [1] * len(ordered)
    for i, (w, h) in enumerate(ordered):
        for j in range(i):
            if ordered[j][0] < w and ordered[j][1] < h:
                dp[i] = max(dp[i], dp[j] + 1)
    return max(dp, default=0)
