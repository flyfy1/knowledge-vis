def subset_baseline(courses):
    ordered = sorted(courses, key=lambda c: c[1])
    best = 0
    for mask in range(1 << len(ordered)):
        total = count = 0
        for i, (duration, deadline) in enumerate(ordered):
            if mask & (1 << i):
                total += duration
                if total > deadline:
                    break
                count += 1
        else:
            best = max(best, count)
    return best
