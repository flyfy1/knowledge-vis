def baseline(nums):
    answer = 0
    for mask in range(1 << len(nums)):
        chosen = [nums[i] for i in range(len(nums)) if mask & (1 << i)]
        if len(chosen) >= 3:
            d = chosen[1] - chosen[0]
            answer += all(chosen[k] - chosen[k-1] == d
                          for k in range(2, len(chosen)))
    return answer
