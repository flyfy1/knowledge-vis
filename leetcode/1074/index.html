def direct(matrix, target):
    m, n = len(matrix), len(matrix[0])
    answer = 0
    for top in range(m):
        for bottom in range(top, m):
            for left in range(n):
                for right in range(left, n):
                    total = 0
                    for r in range(top, bottom + 1):
                        for c in range(left, right + 1):
                            total += matrix[r][c]
                    answer += int(total == target)
    return answer
