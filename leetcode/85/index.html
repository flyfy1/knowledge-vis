def baseline(matrix):
    R, C = len(matrix), len(matrix[0])
    best = 0
    for top in range(R):
        for bottom in range(top, R):
            for left in range(C):
                for right in range(left, C):
                    valid = True
                    for r in range(top, bottom + 1):
                        for c in range(left, right + 1):
                            if matrix[r][c] != "1":
                                valid = False
                    if valid:
                        best = max(best, (bottom-top+1)*(right-left+1))
    return best
