def baseline(s, k):
    seen = {s}
    queue = [s]
    head = 0
    while head < len(queue):
        current = queue[head]
        head += 1
        for j in range(k):
            nxt = current[:j] + current[j + 1:] + current[j]
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return min(seen)
