from itertools import permutations
from math import isqrt

def baseline(nums):
    unique_orders = set(permutations(nums))
    answer = 0
    for order in unique_orders:
        if all(isqrt(a + b) ** 2 == a + b
               for a, b in zip(order, order[1:])):
            answer += 1
    return answer
