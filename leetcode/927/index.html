def baseline(arr):
    def value(bits):
        result = 0
        for bit in bits:
            result = result * 2 + bit
        return result

    n = len(arr)
    for i in range(n - 2):
        for j in range(i + 2, n):
            a = value(arr[:i + 1])
            b = value(arr[i + 1:j])
            c = value(arr[j:])
            if a == b == c:
                return [i, j]
    return [-1, -1]
