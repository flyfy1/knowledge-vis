    from collections import deque
from itertools import product


def slow_min_moves(machines):
    n, total = len(machines), sum(machines)
    if total % n:
        return -1
    goal = (total // n,) * n
    start = tuple(machines)
    queue, seen = deque([(start, 0)]), {start}
    while queue:
        state, moves = queue.popleft()
        if state == goal:
            return moves
        options = []
        for i, count in enumerate(state):
            directions = [0]  # 0: stay; -1: left; +1: right
            if count and i > 0:
                directions.append(-1)
            if count and i + 1 < n:
                directions.append(1)
            options.append(directions)
        for actions in product(*options):
            after = list(state)
            for i, direction in enumerate(actions):
                if direction:
                    after[i] -= 1
                    after[i + direction] += 1
            nxt = tuple(after)
            if nxt not in seen:
                seen.add(nxt)
                queue.append((nxt, moves + 1))
