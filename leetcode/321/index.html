def exhaustive(a, b, k):
    if k == 0:
        return []
    if len(a) + len(b) < k:
        return None
    candidates = []
    for nums, other in ((a, b), (b, a)):
        for p, digit in enumerate(nums):
            tail = exhaustive(nums[p + 1:], other, k - 1)
            if tail is not None:
                candidates.append([digit] + tail)
    return max(candidates, default=None)
