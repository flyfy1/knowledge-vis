    def reaching_points_slow(sx, sy, tx, ty):
    while tx >= sx and ty >= sy:
        if (tx, ty) == (sx, sy):
            return True
        if tx == ty:  # diagonal: no parent
            return False
        if tx > ty:
            tx -= ty
        else:
            ty -= tx
    return False
