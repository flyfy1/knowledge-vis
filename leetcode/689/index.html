def baseline(nums, k):
    m = len(nums) - k + 1
    w = [sum(nums[i:i+k]) for i in range(m)]
    best_sum, answer = -1, None
    for a in range(m):
        for b in range(a+k, m):
            for c in range(b+k, m):
                triple = [a, b, c]
                total = w[a] + w[b] + w[c]
                if (total > best_sum or
                    (total == best_sum and triple < answer)):
                    best_sum, answer = total, triple
    return answer
