from typing import List

def brute(nums: List[int]) -> int:
    answer = 0
    for left in range(len(nums)):
        seen = set()
        for right in range(left, len(nums)):
            seen.add(nums[right])
            answer += len(seen) ** 2
    return answer % (10**9 + 7)
