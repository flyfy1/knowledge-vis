def baseline(stones):
    positions = set(stones)
    def walk(x, k):
        if x == stones[-1]:
            return True
        for jump in (k - 1, k, k + 1):
            if jump > 0 and x + jump in positions:
                if walk(x + jump, jump):
                    return True
        return False
    return walk(0, 0)
