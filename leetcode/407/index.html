def layer_baseline(heightMap):
    m, n = len(heightMap), len(heightMap[0])
    total = 0
    for z in range(1, max(map(max, heightMap)) + 1):
        wet = {(r, c) for r in range(m) for c in range(n)
               if heightMap[r][c] < z}
        outside = {v for v in wet if v[0] in (0, m - 1)
                   or v[1] in (0, n - 1)}
        stack = list(outside)
        while stack:
            r, c = stack.pop()
            for dr, dc in ((-1, 0), (0, -1), (0, 1), (1, 0)):
                v = (r + dr, c + dc)
                if v in wet and v not in outside:
                    outside.add(v)
                    stack.append(v)
        total += len(wet - outside)
    return total
