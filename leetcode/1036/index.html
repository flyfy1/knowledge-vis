from collections import deque

def baseline(blocked, source, target):
    walls = {tuple(p) for p in blocked}
    seen = {tuple(source)}
    queue = deque([tuple(source)])
    while queue:
        x, y = queue.popleft()
        if (x, y) == tuple(target):
            return True
        for dx, dy in ((1,0), (0,1), (-1,0), (0,-1)):
            cell = (x+dx, y+dy)
            if (0 <= cell[0] < 10**6 and 0 <= cell[1] < 10**6
                    and cell not in walls and cell not in seen):
                seen.add(cell)
                queue.append(cell)
    return False
