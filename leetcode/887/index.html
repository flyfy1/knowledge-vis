def baseline(k, n):
    D = [[0] * (n + 1) for _ in range(k + 1)]
    for h in range(1, n + 1):
        D[1][h] = h
    for e in range(2, k + 1):
        for h in range(1, n + 1):
            D[e][h] = 1 + min(
                max(D[e - 1][x - 1], D[e][h - x])
                for x in range(1, h + 1)
            )
    return D[k][n]
