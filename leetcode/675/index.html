def baseline(forest):
    m, n = len(forest), len(forest[0])
    size, inf = m * n, float("inf")
    d = [[inf] * size for _ in range(size)]
    for r in range(m):
        for c in range(n):
            if forest[r][c] == 0:
                continue
            a = r * n + c
            d[a][a] = 0
            for s, t in ((r-1,c), (r+1,c), (r,c-1), (r,c+1)):
                if 0 <= s < m and 0 <= t < n and forest[s][t] > 0:
                    d[a][s*n+t] = 1
    for k in range(size):
        for a in range(size):
            for b in range(size):
                d[a][b] = min(d[a][b], d[a][k] + d[k][b])
    trees = sorted((forest[r][c], r*n+c)
                   for r in range(m) for c in range(n)
                   if forest[r][c] > 1)
    current = total = 0
    for _, target in trees:
        if d[current][target] == inf:
            return -1
        total += d[current][target]
        current = target
    return total
