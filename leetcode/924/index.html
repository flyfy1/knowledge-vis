def baseline(graph, initial):
    answer, best = min(initial), len(graph) + 1
    for removed in sorted(initial):
        seen = set(initial) - {removed}
        queue = list(seen)
        head = 0
        while head < len(queue):
            node = queue[head]
            head += 1
            for neighbor in range(len(graph)):
                if graph[node][neighbor] and neighbor not in seen:
                    seen.add(neighbor)
                    queue.append(neighbor)
        if len(seen) < best:
            answer, best = removed, len(seen)
    return answer
