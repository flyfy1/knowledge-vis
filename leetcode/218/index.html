def baseline(buildings):
    xs = sorted({x for l, r, h in buildings for x in (l, r)})
    result = []
    for x in xs:
        h = max((h for l, r, h in buildings if l <= x < r), default=0)
        if not result or result[-1][1] != h:
            result.append([x, h])
    return result
