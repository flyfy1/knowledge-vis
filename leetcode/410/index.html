from itertools import combinations

def baseline(nums, k):
    best = sum(nums)
    for cuts in combinations(range(1, len(nums)), k - 1):
        ends = (0,) + cuts + (len(nums),)
        largest = max(sum(nums[a:b])
                      for a, b in zip(ends, ends[1:]))
        best = min(best, largest)
    return best
