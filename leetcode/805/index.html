def baseline(nums):
    n, total = len(nums), sum(nums)
    for mask in range(1, (1 << n) - 1):
        k = s = 0
        for i, x in enumerate(nums):
            if mask & (1 << i):
                k += 1
                s += x
        if n * s == k * total:
            return True
    return False
