def baseline(dungeon):
    m, n = len(dungeon), len(dungeon[0])
    def visit(r, c, s, low):
        s += dungeon[r][c]
        low = min(low, s)
        if r == m - 1 and c == n - 1:
            return 1 - low
        choices = []
        if r + 1 < m:
            choices.append(visit(r + 1, c, s, low))
        if c + 1 < n:
            choices.append(visit(r, c + 1, s, low))
        return min(choices)
    return visit(0, 0, 0, 0)
