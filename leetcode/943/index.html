from itertools import permutations

def baseline(words):
    best = None
    for order in permutations(words):
        result = order[0]
        for word in order[1:]:
            k = min(len(result), len(word))
            while k and result[-k:] != word[:k]:
                k -= 1
            result += word[k:]
        if best is None or len(result) < len(best):
            best = result
    return best
