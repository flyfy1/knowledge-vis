def baseline(graph):
    n = len(graph)
    paths = [[u] for u in range(n)]
    for distance in range(2 * (n - 1) + 1):
        following = []
        for path in paths:
            if len(set(path)) == n:
                return distance
            if distance < 2 * (n - 1):
                for v in graph[path[-1]]:
                    following.append(path + [v])
        paths = following
