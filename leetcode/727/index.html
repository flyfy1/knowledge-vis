def brute(s1, s2):
    best = ""
    for i in range(len(s1)):              # every start
        k = 0                             # letters of s2 matched so far
        for j in range(i, len(s1)):
            if s1[j] == s2[k]:
                k += 1
                if k == len(s2):          # s1[i..j]: shortest window from i
                    if best == "" or j - i + 1 < len(best):
                        best = s1[i:j + 1]
                    break
    return best
