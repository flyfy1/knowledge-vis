def baseline(strs):
    m = len(strs[0])
    choices = [[]]
    for j in range(m):
        choices += [kept + [j] for kept in choices]
    best = 0
    for kept in choices:
        valid = True
        for row in strs:
            for k in range(1, len(kept)):
                if row[kept[k-1]] > row[kept[k]]:
                    valid = False
        if valid:
            best = max(best, len(kept))
    return m - best
