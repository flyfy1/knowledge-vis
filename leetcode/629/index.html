    from itertools import permutations

def brute(n, k):
    answer = 0
    for p in permutations(range(1, n + 1)):
        inversions = sum(p[a] > p[b]
                         for a in range(n)
                         for b in range(a + 1, n))
        answer += inversions == k
    return answer % 1_000_000_007
